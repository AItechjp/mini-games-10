onNetwork(m){if(!m||m.from===state.playerId)return;switch(m.kind){case'hello':if(state.role==='host'){state.partnerReady=true;state.players.set(m.from,{id:m.from,x:0,z:14,yaw:0,lives:3,shield:0});send('settings',{difficulty:state.difficulty,hostId:state.playerId});updateLobby();}else if(m.role==='host'){state.hostId=m.from;state.partnerReady=true;updateLobby();}break;case'settings':if(state.role==='guest'){state.difficulty=m.difficulty||state.difficulty;state.hostId=m.hostId||m.from;syncDifficultyUI();state.partnerReady=true;updateLobby();}break;case'start':if(state.role==='guest'){state.area=m.area||0;state.seed=m.seed>>>0;state.difficulty=m.difficulty||'normal';state.hostId=m.hostId||m.from;state.localLives=3;state.shield=0;state.players.clear();for(const[id,lives,shield=0]of m.lives||[])state.players.set(id,{id,x:0,z:14,yaw:0,lives,shield});if(!state.players.has(state.playerId))state.players.set(state.playerId,{id:state.playerId,x:0,z:14,yaw:0,lives:3,shield:0});syncDifficultyUI();state.startedAt=performance.now();beginArea(false);}break;case'player':if(state.role==='host'&&m.area===state.area){const p=state.players.get(m.from)||{id:m.from,lives:3,shield:0};Object.assign(p,{x:m.x,z:m.z,yaw:m.yaw});state.players.set(m.from,p);}break;case'world':if(state.role==='guest'&&m.area===state.area)syncWorld(m);break;case'hit':if(state.role==='host'&&m.area===state.area)hostHit(m.enemyId,m.playerId||m.from);break;case'enemy':if(state.role==='guest'&&m.area===state.area){const e=state.enemies.get(m.id);if(e){e.hp=m.hp;e.dead=m.dead;}if(m.id==='boss'&&bossMesh)bossMesh.visible=!m.dead;updateBossHud();}break;case'damage':if(m.target===state.playerId&&m.area===state.area){state.localLives=m.lives;state.shield=m.shield||0;const p=state.players.get(state.playerId)||{id:state.playerId};Object.assign(p,{lives:m.lives,shield:m.shield||0});state.players.set(state.playerId,p);if(m.blocked)toast('ANTIVIRAL SHIELD');else flashDamage();updateHud();}break;case'item':if(m.area===state.area){const item=state.items.get(m.id);if(item){item.taken=true;if(item.mesh)item.mesh.visible=false;}const p=state.players.get(m.target)||{id:m.target,lives:m.lives,shield:m.shield||0};p.lives=m.lives;p.shield=m.shield||0;state.players.set(m.target,p);if(m.target===state.playerId){state.localLives=m.lives;state.shield=m.shield||0;if(m.type==='adrenaline')state.boostUntil=performance.now()+10000;toast(m.type==='medkit'?'MEDKIT +1 LIFE':m.type==='shield'?'ANTIVIRAL SHIELD':'ADRENALINE 10s',1700);updateHud();}}break;case'area_clear':if(state.role==='guest'&&m.area===state.area){state.running=false;toast(`AREA ${state.area+1} CLEAR`,1900);}break;case'area_start':if(state.role==='guest'){state.area=m.area;state.seed=m.seed>>>0;state.difficulty=m.difficulty||state.difficulty;for(const[id,lives,shield=0]of m.lives||[]){const p=state.players.get(id)||{id,x:0,z:14,yaw:0};p.lives=lives;p.shield=shield;state.players.set(id,p);if(id===state.playerId){state.localLives=lives;state.shield=shield;}}beginArea(false);}break;case'fail':if(state.role==='guest'){state.running=false;setOverlay('MISSION FAILED',m.reason||'チームが倒れた。','ホストの再開を待つ',true);startBtn.disabled=true;}break;case'complete':if(state.role==='guest'){state.running=false;state.completed=true;setOverlay('BLACK SITE ESCAPED',`5区域を協力で突破。総時間 ${fmt(m.elapsed||0)}。`,'完了',true);startBtn.disabled=true;}break;}}
function sendWorld(force=false){if(!isHost()||state.mode!=='coop'||!state.connected||!state.running)return;const now=performance.now();if(!force&&now-state.lastSnapshot<170)return;state.lastSnapshot=now;send('world',{area:state.area,enemies:[...state.enemies.values()].map(e=>[e.id,e.index??-1,+e.x.toFixed(2),+e.z.toFixed(2),e.hp,e.dead?1:0,e.boss?1:0,e.maxHp,+e.speed.toFixed(2),e.type,e.phase||0,e.lean||0,e.headScale||1,e.visualYaw||0,e.drag?1:0,e.twist||0,e.headTilt||0,e.headYaw||0,e.headSide||0,e.armDrop||0,e.bodyScale||1,e.shoulderSide||0,e.shoulderScale||1]),players:[...state.players.values()].map(p=>[p.id,+p.x.toFixed(2),+p.z.toFixed(2),+p.yaw.toFixed(3),p.lives,p.shield||0]),items:[...state.items.values()].map(i=>[i.id,i.taken?1:0])});}
function applyWorldEnemy(a){const[id,index,x,z,hp,dead,boss,maxHp,speed,type,phase,lean,headScale,visualYaw,drag,twist,headTilt,headYaw,headSide,armDrop,bodyScale,shoulderSide,shoulderScale]=a;let e=state.enemies.get(id);if(!e){e={id,index,x,z,hp,dead:!!dead,boss:!!boss,maxHp,speed,type,phase,lean,headScale,visualYaw,drag:!!drag,twist,headTilt,headYaw,headSide,armDrop,bodyScale,shoulderSide,shoulderScale};state.enemies.set(id,e);if(e.boss&&!bossMesh)createBossMesh(e);}else Object.assign(e,{index,x,z,hp,dead:!!dead,boss:!!boss,maxHp,speed,type,phase,lean,headScale,visualYaw,drag:!!drag,twist,headTilt,headYaw,headSide,armDrop,bodyScale,shoulderSide,shoulderScale});}
function syncWorld(m){const normal=(m.enemies||[]).filter(a=>!a[6]);if(!horde&&normal.length)makeHorde(Math.max(normal.length,difficulty().count[state.area]));for(const a of m.enemies||[])applyWorldEnemy(a);syncRemotePlayers(m.players||[]);for(const p of m.players||[])if(p[0]===state.playerId){const me=state.players.get(state.playerId)||{id:state.playerId};me.lives=p[4];me.shield=p[5]||0;state.players.set(state.playerId,me);state.localLives=p[4];state.shield=p[5]||0;}for(const [id,taken]of m.items||[]){const it=state.items.get(id);if(it){it.taken=!!taken;if(it.mesh)it.mesh.visible=!it.taken;}}updateBossHud();updateHud();}
function networkTick(now){if(state.mode!=='coop'||!state.connected||!state.running)return;if(!isHost()&&now-state.lastPlayerSend>75){state.lastPlayerSend=now;send('player',{area:state.area,x:+local.x.toFixed(2),z:+local.z.toFixed(2),yaw:+local.yaw.toFixed(3)});}if(isHost())sendWorld(false);}
